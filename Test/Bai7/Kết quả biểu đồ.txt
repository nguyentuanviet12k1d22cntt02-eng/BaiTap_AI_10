function khoiTaoDashboard() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var requiredSheets = ['DanhMuc_BT7', 'SanPham_BT7', 'KhachHang_BT7', 'DonHang_BT7', 'ChiTietDonHang_BT7'];
  for (var i = 0; i < requiredSheets.length; i++) {
    if (!ss.getSheetByName(requiredSheets[i])) {
      var errorMsg = 'Thiếu trang tính nguồn: "' + requiredSheets[i] + '". Vui lòng kiểm tra lại cấu trúc bảng tính!';
      Logger.log(errorMsg);
      try {
        SpreadsheetApp.getUi().alert('Lỗi Dữ Liệu', errorMsg, SpreadsheetApp.getUi().ButtonSet.OK);
      } catch (e) {}
      return;
    }
  }

  try {
    var calcSheet = taoHoacLamMoiSheet(ss, 'Calc_Data');
    thietLapTrangCalcData(calcSheet);

    var dashSheet = taoHoacLamMoiSheet(ss, '📊 Dashboard');
    ss.setActiveSheet(dashSheet);
    ss.moveActiveSheet(1);

    thietLapGiaoDienDashboard(dashSheet);

    SpreadsheetApp.flush();

    veBieuDoDashboard(dashSheet, calcSheet);

    try {
      SpreadsheetApp.getUi().alert(
        'Thành Công',
        'Đã cập nhật và làm mới Dashboard Tổng Quan thành công!',
        SpreadsheetApp.getUi().ButtonSet.OK
      );
    } catch (e) {}

  } catch (err) {
    Logger.log('Lỗi trong quá trình khởi tạo Dashboard: ' + err.toString());
    try {
      SpreadsheetApp.getUi().alert('Lỗi', 'Không thể khởi tạo Dashboard: ' + err.message, SpreadsheetApp.getUi().ButtonSet.OK);
    } catch (e) {}
  }
}

function taoHoacLamMoiSheet(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (sheet) {
    sheet.clear();
    sheet.clearNotes();
    var charts = sheet.getCharts();
    for (var i = 0; i < charts.length; i++) {
      sheet.removeChart(charts[i]);
    }
  } else {
    sheet = ss.insertSheet(sheetName);
  }
  return sheet;
}

function thietLapTrangCalcData(sheet) {
  sheet.getRange('A1:B1').setValues([['Tên Danh Mục', 'Doanh Thu']]).setFontWeight('bold');
  
  var categoryFormulas = [];
  for (var r = 4; r <= 10; r++) {
    categoryFormulas.push([
      '=DanhMuc_BT7!B' + r,
      '=SUM(IFERROR(FILTER(ChiTietDonHang_BT7!H$4:H; ISNUMBER(MATCH(ChiTietDonHang_BT7!C$4:C; FILTER(SanPham_BT7!A$4:A; SanPham_BT7!C$4:C=DanhMuc_BT7!A' + r + '); 0))); 0))'
    ]);
  }
  sheet.getRange(2, 1, categoryFormulas.length, 2).setFormulas(categoryFormulas);
  sheet.getRange('B2:B8').setNumberFormat('#,##0');

  var top10QueryFormula = '=QUERY(ChiTietDonHang_BT7!D4:F; "SELECT D, SUM(F) WHERE D IS NOT NULL GROUP BY D ORDER BY SUM(F) DESC LIMIT 10 LABEL D \'Tên Sản Phẩm\', SUM(F) \'Số Lượng Đã Bán\'")';
  sheet.getRange('D1').setFormula(top10QueryFormula);

  SpreadsheetApp.flush();
}

function veBieuDoDashboard(dashSheet, calcSheet) {
  var pieRange = calcSheet.getRange('A1:B8');
  var pieChart = dashSheet.newChart()
    .setChartType(Charts.ChartType.PIE)
    .addRange(pieRange)
    .setNumHeaders(1)
    .setPosition(9, 1, 0, 0)
    .setOption('title', '📊 TỶ LỆ DOANH THU THEO DANH MỤC')
    .setOption('titleTextStyle', { fontSize: 13, bold: true, color: '#0f4c81' })
    .setOption('pieSliceText', 'percentage')
    .setOption('legend', { position: 'right', textStyle: { fontSize: 11, color: '#334e68' } })
    .setOption('width', 490)
    .setOption('height', 360)
    .build();
  dashSheet.insertChart(pieChart);

  var colRange = calcSheet.getRange('D1:E11');
  var colChart = dashSheet.newChart()
    .setChartType(Charts.ChartType.COLUMN)
    .addRange(colRange)
    .setNumHeaders(1)
    .setPosition(9, 5, 0, 0)
    .setOption('title', '🏆 TOP 10 SẢN PHẨM BÁN CHẠY NHẤT')
    .setOption('titleTextStyle', { fontSize: 13, bold: true, color: '#0f4c81' })
    .setOption('colors', ['#1a73e8'])
    .setOption('legend', { position: 'none' })
    .setOption('hAxis', {
      title: 'Tên Sản Phẩm',
      slantedText: true,
      slantedTextAngle: 30,
      textStyle: { fontSize: 9, color: '#334e68' }
    })
    .setOption('vAxis', {
      title: 'Số Lượng Đã Bán',
      minValue: 0,
      textStyle: { fontSize: 10, color: '#334e68' }
    })
    .setOption('width', 560)
    .setOption('height', 360)
    .build();
  dashSheet.insertChart(colChart);
}
